1. database/sql 與 driver

標準庫的 database/sql 只定義介面，不認得任何一種資料庫。driver 套件在自己的 init 裡呼叫 sql.Register("sqlite", …) 註冊一個名字，程式用空白 import 讓 init 執行，再用 sql.Open("sqlite", 連線字串) 選它：

func openDB(ctx context.Context, path string) (*sql.DB, error) {
	// ? 後面底線開頭的參數由 driver 處理：開啟外鍵檢查、被鎖住時最多等 5 秒、
	// 使用 WAL 模式（讀寫可以同時進行）、交易一開始就拿寫入鎖（第 6 節）。
	dsn := path + "?_foreign_keys=1&_busy_timeout=5000&_journal_mode=WAL&_txlock=immediate"
	db, err := sql.Open("sqlite", dsn)
	if err != nil {
		return nil, err
	}
	db.SetMaxOpenConns(8)
	db.SetMaxIdleConns(8)
	db.SetConnMaxIdleTime(5 * time.Minute)

	if err := db.PingContext(ctx); err != nil {
		db.Close()
		return nil, fmt.Errorf("連不上資料庫：%w", err)
	}
	if err := migrate(ctx, db); err != nil {
		db.Close()
		return nil, err
	}
	return db, nil
}
func openDemo() {
	ctx := context.Background()
	dir, err := os.MkdirTemp("", "bento-*")
	if err != nil {
		fmt.Println(err)
		return
	}

  • sql.Open 不會真的連線，只檢查參數、準備好連線池。第一次需要連線時才連，所以開啟之後用 PingContext 確認連得上，有問題在啟動時就發現。
  • *sql.DB 是一個連線池，可以被很多 goroutine 同時使用。整個程式開一個，程式結束時 Close；不要每個請求開一次。
  • 連線字串的格式是 driver 自己定的。這裡 ? 後面底線開頭的參數，modernc.org/sqlite 會轉成 SQLite 的 PRAGMA 設定，例如 SQLite 預設不檢查外鍵，要自己打開。
  • openDB 開啟失敗時 Close 已經開的資源再回傳錯誤，呼叫的一方拿到錯誤就不用再收拾。

2. 寫入：Exec

不回傳資料列的 SQL 用 ExecContext。值用 ? 帶入（第 4 節說明為什麼），回傳的 sql.Result 告訴你新增的 id 和改到幾列。把所有的 SQL 集中在一個 store 型別裡，其他程式只看到「新增一道菜」這種方法：

var (
	ErrNotFound  = errors.New("找不到")
	ErrDuplicate = errors.New("已經有了")
	ErrSoldOut   = errors.New("賣完了")
)

type Dish struct {
	ID    int64
	Name  string
	Price int
	Stock int
}

type store struct {
	db *sql.DB
}

func isUnique(err error) bool {
	e, ok := errors.AsType[*sqlite.Error](err)
	return ok && e.Code() == sqlite3.SQLITE_CONSTRAINT_UNIQUE
}

func (s *store) addDish(ctx context.Context, name string, price, stock int) (int64, error) {
	res, err := s.db.ExecContext(ctx,
		`INSERT INTO dishes (name, price, stock) VALUES (?, ?, ?)`, name, price, stock)
	if isUnique(err) {
		return 0, fmt.Errorf("%s %w", name, ErrDuplicate)
	}
	if err != nil {
		return 0, err
	}
	return res.LastInsertId()
}

func (s *store) restock(ctx context.Context, name string, n int) error {
	res, err := s.db.ExecContext(ctx,
		`UPDATE dishes SET stock = stock + ? WHERE name = ?`, n, name)
	if err != nil {
		return err
	}
	changed, err := res.RowsAffected()
	if err != nil {
		return err
	}
	if changed == 0 {
		return fmt.Errorf("%s：%w", name, ErrNotFound)
	}
	return nil
}
func execDemo() {
	ctx := context.Background()
	s, done := newDemoStore(false)
	defer done()

	for _, d := range []Dish{{Name: "滷肉飯", Price: 50, Stock: 20}, {Name: "排骨飯", Price: 90, Stock: 5}, {Name: "雞腿飯", Price: 100, Stock: 0}} {
		id, err := s.addDish(ctx, d.Name, d.Price, d.Stock)
		fmt.Println("新增", d.Name, "id =", id, err)
	}

	_, err := s.addDish(ctx, "排骨飯", 95, 3)
	fmt.Println(err, errors.Is(err, ErrDuplicate))
	_, err = s.addDish(ctx, "空氣便當", 0, 1)
	fmt.Println(err)

	fmt.Printf("補雞腿飯：%v\n", s.restock(ctx, "雞腿飯", 10))
	fmt.Printf("補鰻魚飯：%v\n", s.restock(ctx, "鰻魚飯", 10))
}

  • 違反 UNIQUE 是「名稱重複」，呼叫的一方需要知道；store 把 driver 的錯誤（*sqlite.Error 和它的代碼）轉成自己的 ErrDuplicate。driver 的型別只出現在 isUnique 裡，換資料庫時只改這裡。
  • 價格 0 這種錯誤是程式的 bug，不需要特別轉換，原樣回傳就好；呼叫的一方會記錄下來。
  • UPDATE 找不到符合的列，資料庫不會報錯，只是改到 0 列。restock 用 RowsAffected 判斷，轉成 ErrNotFound。
  • LastInsertId 不是每種資料庫都支援。PostgreSQL 的 driver 要用 INSERT … RETURNING id 搭配 QueryRow 拿新的 id。
  • 每個方法的第一個參數都是 ctx，並使用 …Context 版本的方法：請求被取消時，查詢也跟著停（第 7 節）。

3. 讀取：Query 與 QueryRow

回傳多列用 QueryContext，然後用 Next、Scan 一列一列讀；只要一列用 QueryRowContext，它的錯誤延到 Scan 才回傳：

func (s *store) findDish(ctx context.Context, name string) (Dish, error) {
	var d Dish
	err := s.db.QueryRowContext(ctx,
		`SELECT id, name, price, stock FROM dishes WHERE name = ?`, name,
	).Scan(&d.ID, &d.Name, &d.Price, &d.Stock)
	if errors.Is(err, sql.ErrNoRows) {
		return Dish{}, fmt.Errorf("%s：%w", name, ErrNotFound)
	}
	return d, err
}

func (s *store) listDishes(ctx context.Context, maxPrice int) ([]Dish, error) {
	rows, err := s.db.QueryContext(ctx,
		`SELECT id, name, price, stock FROM dishes WHERE price <= ? ORDER BY price, id`, maxPrice)
	if err != nil {
		return nil, err
	}
	defer rows.Close()

	var dishes []Dish
	for rows.Next() {
		var d Dish
		if err := rows.Scan(&d.ID, &d.Name, &d.Price, &d.Stock); err != nil {
			return nil, err
		}
		dishes = append(dishes, d)
	}
	return dishes, rows.Err()
}
func queryDemo() {
	ctx := context.Background()
	s, done := newDemoStore(true)
	defer done()

	dishes, err := s.listDishes(ctx, 90)
	if err != nil {
		fmt.Println(err)
		return
	}
	for _, d := range dishes {
		fmt.Printf("%d %s %d 元，剩 %d 份\n", d.ID, d.Name, d.Price, d.Stock)
	}

	d, err := s.findDish(ctx, "雞腿飯")
	fmt.Printf("%+v %v\n", d, err)
	_, err = s.findDish(ctx, "鰻魚飯")
	fmt.Println(err, errors.Is(err, ErrNotFound))
}

2 排骨飯 90 元，剩 5 份
{ID:3 Name:雞腿飯 Price:100 Stock:0} <nil>

  • Scan 的順序和 SELECT 的欄位順序一一對應，個數也要一樣。所以永遠寫出欄位名稱，不要 SELECT *：資料表加了一個欄位，Scan 就壞了。
  • rows.Err()：Next 回傳 false 可能是讀完了，也可能是讀到一半連線斷了。沒檢查的話，只讀到一半的結果會被當成完整的。
  • sql.ErrNoRows 只會從 QueryRow 的 Scan 回傳。查不到常常是正常情況，store 把它轉成自己的 ErrNotFound，呼叫的一方用 errors.Is 判斷（11 章）。Query 查不到則是迴圈一次都沒有跑，不是錯誤。
  • ORDER BY 寫清楚。沒有 ORDER BY 時，資料庫回傳的順序沒有保證，今天剛好是新增的順序，加了索引之後可能就變了。

4. 參數與 SQL injection

SQL injection 是最常見的資料庫漏洞：程式把使用者的輸入拼進 SQL，輸入就能改變 SQL 的意思。比較兩種寫法：

func (s *store) names(ctx context.Context, query string, args ...any) ([]string, error) {
	rows, err := s.db.QueryContext(ctx, query, args...)
	if err != nil {
		return nil, err
	}
	defer rows.Close()

	var out []string
	for rows.Next() {
		var name string
		if err := rows.Scan(&name); err != nil {
			return nil, err
		}
		out = append(out, name)
	}
	return out, rows.Err()
}
func (s *store) searchUnsafe(ctx context.Context, name string) ([]string, error) {
	return s.names(ctx, "SELECT name FROM dishes WHERE name = '"+name+"'")
}

func (s *store) search(ctx context.Context, name string) ([]string, error) {
	return s.names(ctx, "SELECT name FROM dishes WHERE name = ?", name)
}

var sortColumns = map[string]string{
	"price": "price, id",
	"name":  "name",
	"stock": "stock DESC, id",
}

func (s *store) dishNamesBy(ctx context.Context, sort string) ([]string, error) {
	order, ok := sortColumns[sort]
	if !ok {
		return nil, fmt.Errorf("不能用 %q 排序", sort)
	}
	return s.names(ctx, "SELECT name FROM dishes ORDER BY "+order)
}

func injectionDemo() {
	ctx := context.Background()
	s, done := newDemoStore(true)
	defer done()

	for _, input := range []string{"排骨飯", "x' OR '1'='1", "阿嬤's 滷肉"} {
		got, err := s.searchUnsafe(ctx, input)
		fmt.Printf("拼字串 %q → %v %v\n", input, got, err)
		got, err = s.search(ctx, input)
		fmt.Printf("用 ?   %q → %v %v\n", input, got, err)
	}

	for _, sort := range []string{"stock", "price; DROP TABLE dishes"} {
		got, err := s.dishNamesBy(ctx, sort)
		fmt.Println("排序", sort, "→", got, err)
	}
}

用 ?   "排骨飯" → [排骨飯] <nil>
拼字串 "x' OR '1'='1" → [排骨飯 滷肉飯 雞腿飯] <nil>

用 ?   "阿嬤's 滷肉" → [] <nil>

排序 price; DROP TABLE dishes → [] 不能用 "price; DROP TABLE dishes" 排序
  • 拼字串的版本，一般的輸入看起來完全正常，測試也會過。輸入裡有單引號時，客人的名字「阿嬤's 滷肉」讓 SQL 壞掉；攻擊者的輸入則讓它查出所有資料，甚至改資料。
  • 用 ? 的版本，三種輸入都只是要找的名稱。不管輸入是什麼，都不需要額外處理。
  • 參數的寫法各家不同：SQLite、MySQL 用 ?，PostgreSQL 用 $1、$2。
  • 參數只能代替「值」，不能代替資料表名稱、欄位名稱、ASC/DESC。這些需要依輸入改變時，像 sortColumns 那樣只從固定的清單裡挑，清單外的一律拒絕。
  • LIKE 的 % 和 _ 是萬用字元。用 ? 帶入 "%" + 輸入 + "%" 不會被 injection，但使用者輸入的 % 也會被當成萬用字元。

5. NULL

訂單的備註可以不寫，資料表裡就是 NULL。直接 Scan 進 string 會失敗；Go 1.22 起標準庫有泛型的 sql.Null[T]，用 Valid 分辨「有值」和「NULL」：

type Order struct {
	ID     int64
	DishID int64
	Qty    int
	Total  int
	Note   sql.Null[string]
}
func nullDemo() {
	ctx := context.Background()
	s, done := newDemoStore(true)
	defer done()

	_, err := s.db.ExecContext(ctx, `INSERT INTO orders (dish_id, qty, total, note)
		VALUES (1, 1, 50, NULL), (2, 2, 180, '不要辣')`)
	if err != nil {
		fmt.Println(err)
		return
	}

	var note string
	err = s.db.QueryRowContext(ctx, `SELECT note FROM orders WHERE id = 1`).Scan(&note)
	fmt.Printf("Scan 進 string：%v\n", err)

	for id := 1; id <= 2; id++ {
		var o Order
		err := s.db.QueryRowContext(ctx,
			`SELECT id, note FROM orders WHERE id = ?`, id).Scan(&o.ID, &o.Note)
		if err != nil {
			fmt.Println(err)
			return
		}
		if o.Note.Valid {
			fmt.Printf("訂單 %d 的備註：%s\n", o.ID, o.Note.V)
		} else {
			fmt.Printf("訂單 %d 沒有備註\n", o.ID)
		}
	}

	notes, err := s.names(ctx, `SELECT COALESCE(note, '（無）') FROM orders ORDER BY id`)
	fmt.Printf("用 COALESCE：%v %v\n", notes, err)
}
Scan 進 string：sql: Scan error on column index 0, name "note": converting NULL to string is unsupported
訂單 1 沒有備註
訂單 2 的備註：不要辣
用 COALESCE：[（無） 不要辣] <nil>
  • 寫入 NULL：參數傳 Valid 是 false 的 sql.Null，或直接傳 nil。第 6 節的 placeOrder 沒有備註時就存 NULL。
  • 指標也可以：Scan 進 *string 型別的變數，NULL 會變成 nil。在 JSON 裡 nil 會變成 null，有時比 sql.Null 方便（05 章）。
  • Go 1.22 以前的程式會看到 sql.NullString、sql.NullInt64 這些各別的型別，意思一樣。
  • 設計資料表時，能用 NOT NULL 加上預設值的欄位就這樣做，程式就不用到處處理 NULL。只有「沒有值」本身有意義的欄位才允許 NULL，例如 ISBN 不明的舊書（練習 1）。
  • NULL 在 SQL 裡也很特別：note = NULL 永遠不成立，要寫 note IS NULL；UNIQUE 允許很多列都是 NULL。

6. 交易

下單要做三件事：確認庫存、扣庫存、新增訂單。扣了庫存之後新增訂單失敗，就會少一份便當卻沒有訂單。把它們放進同一個交易，要嘛全部生效，要嘛全部復原：

func (s *store) placeOrder(ctx context.Context, dish string, qty int, note string) (Order, error) {
	tx, err := s.db.BeginTx(ctx, nil)
	if err != nil {
		return Order{}, err
	}
	defer tx.Rollback() // 中途 return 時復原；Commit 之後呼叫則什麼都不做

	var id int64
	var price, stock int
	err = tx.QueryRowContext(ctx,
		`SELECT id, price, stock FROM dishes WHERE name = ?`, dish).Scan(&id, &price, &stock)
	if errors.Is(err, sql.ErrNoRows) {
		return Order{}, fmt.Errorf("%s：%w", dish, ErrNotFound)
	}
	if err != nil {
		return Order{}, err
	}
	if stock < qty {
		return Order{}, fmt.Errorf("%s %w，剩 %d 份", dish, ErrSoldOut, stock)
	}

	if _, err := tx.ExecContext(ctx,
		`UPDATE dishes SET stock = stock - ? WHERE id = ?`, qty, id); err != nil {
		return Order{}, err
	}
	o := Order{DishID: id, Qty: qty, Total: price * qty, Note: sql.Null[string]{V: note, Valid: note != ""}}
	res, err := tx.ExecContext(ctx,
		`INSERT INTO orders (dish_id, qty, total, note) VALUES (?, ?, ?, ?)`, o.DishID, o.Qty, o.Total, o.Note)
	if err != nil {
		return Order{}, err
	}
	if o.ID, err = res.LastInsertId(); err != nil {
		return Order{}, err
	}
	return o, tx.Commit()
}
func txDemo() {
	ctx := context.Background()
	s, done := newDemoStore(true)
	defer done()

	o, err := s.placeOrder(ctx, "排骨飯", 2, "不要辣")

	fmt.Println(err, errors.Is(err, ErrSoldOut))
	_, err = s.placeOrder(ctx, "鰻魚飯", 1, "")
	fmt.Println(err)

	d, _ := s.findDish(ctx, "排骨飯")
	var orders int
	s.db.QueryRowContext(ctx, `SELECT count(*) FROM orders`).Scan(&orders)

  • defer tx.Rollback() 寫在 BeginTx 成功之後。任何一條 return 的路都會復原；成功時先 Commit，之後的 Rollback 什麼都不做（它回傳的 ErrTxDone 忽略即可）。
  • 交易裡的查詢一定要用 tx 的方法。不小心用了 s.db，那個查詢會從連線池拿另一條連線，不在這個交易裡：看不到交易裡還沒 Commit 的改變，失敗時也不會跟著復原。
  • 交易要短：交易開著的時候，它鎖住的資料別人要等。不要在交易裡呼叫外部的 API 或等使用者輸入。

同時有很多人下單時會怎樣？測試讓 20 個人同時搶 5 份排骨飯，用 -race 跑：

func newTestStore(t *testing.T) *store {
	t.Helper()
	db, err := openDB(t.Context(), filepath.Join(t.TempDir(), "test.db"))
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { db.Close() })
	return &store{db: db}
}

func TestPlaceOrderConcurrent(t *testing.T) {
	s := newTestStore(t)
	ctx := t.Context()
	if _, err := s.addDish(ctx, "排骨飯", 90, 5); err != nil {
		t.Fatal(err)
	}

	var ok, soldOut atomic.Int64
	var wg sync.WaitGroup
	for range 20 {
		wg.Go(func() {
			_, err := s.placeOrder(ctx, "排骨飯", 1, "")
			switch {
			case err == nil:
				ok.Add(1)
			case errors.Is(err, ErrSoldOut):
				soldOut.Add(1)
			default:
				t.Errorf("非預期的錯誤：%v", err)
			}
		})
	}
	wg.Wait()

	d, err := s.findDish(ctx, "排骨飯")
	if err != nil {
		t.Fatal(err)
	}
	var orders int
	if err := s.db.QueryRowContext(ctx, `SELECT count(*) FROM orders`).Scan(&orders); err != nil {
		t.Fatal(err)
	}
	if ok.Load() != 5 || soldOut.Load() != 15 || d.Stock != 0 || orders != 5 {
		t.Errorf("成功 %d、賣完 %d、剩 %d 份、%d 筆訂單；want 5、15、0、5",
			ok.Load(), soldOut.Load(), d.Stock, orders)
	}
}

深入：第一次讀可以先跳過

為什麼不會超賣：鎖與隔離等級

placeOrder 先讀庫存再扣。如果兩個交易同時讀到「剩 1 份」，兩個都會扣，庫存就變成 -1。會不會發生，取決於資料庫怎麼處理同時進行的交易。

  • SQLite 同一時間只允許一個寫入的交易。openDB 的 _txlock=immediate 讓交易一開始就拿寫入鎖，其他交易在 BeginTx 排隊（最多等 busy_timeout 的 5 秒），輪到它們時已經看得到前一筆的結果。沒有這個設定時，交易先讀再寫，兩個交易可能都讀完了才搶寫入鎖，其中一個會直接收到「database is locked」。
  • MySQL、PostgreSQL 允許很多交易同時寫，預設的隔離等級下，上面的「先讀再扣」會超賣。常見的做法有兩種：讀的時候加鎖（SELECT … FOR UPDATE），或把條件寫進 UPDATE：UPDATE dishes SET stock = stock - ? WHERE id = ? AND stock >= ?，再看 RowsAffected 是不是 1。後者在每一種資料庫都成立。
  • dishes 資料表的 CHECK (stock >= 0) 是最後一道防線：就算程式寫錯，庫存變成負數的 UPDATE 也會失敗，交易跟著復原。
  • BeginTx 的第二個參數 *sql.TxOptions 可以指定隔離等級和唯讀，各家資料庫支援的不同。

7. 連線池與 context

*sql.DB 依需要開新連線，用完放回池子給下一個查詢。上限用 SetMaxOpenConns 設：資料庫能承受的連線數有限，好幾個程式共用同一個資料庫時更要設。為了看清楚，示範把上限縮成 1 條：

func poolDemo() {
	ctx := context.Background()
	s, done := newDemoStore(true)
	defer done()
	s.db.SetMaxOpenConns(1) // 示範用：連線池只有 1 條連線

	rows, err := s.db.QueryContext(ctx, `SELECT name FROM dishes`)
	if err != nil {
		fmt.Println(err)
		return
	}
	// 忘了 rows.Close()：唯一的連線一直被佔著。

	waitCtx, cancel := context.WithTimeout(ctx, 100*time.Millisecond)
	defer cancel()
	_, err = s.findDish(waitCtx, "排骨飯")
	st := s.db.Stats()
	fmt.Printf("下一個查詢：%v\n", err)
	fmt.Printf("使用中 %d 條，等過 %d 次\n", st.InUse, st.WaitCount)

	rows.Close()
	_, err = s.findDish(ctx, "排骨飯")
	fmt.Printf("關掉 rows 之後：%v\n", err)

	slowCtx, cancel := context.WithTimeout(ctx, 100*time.Millisecond)
	defer cancel()
	var n int
	err = s.db.QueryRowContext(slowCtx, `WITH RECURSIVE c(x) AS (
		SELECT 1 UNION ALL SELECT x + 1 FROM c WHERE x < 1000000000
	) SELECT count(*) FROM c`).Scan(&n)
	fmt.Printf("數到十億：%v %v\n", err, errors.Is(err, context.DeadlineExceeded))
}
下一個查詢：context deadline exceeded
使用中 1 條，等過 1 次
關掉 rows 之後：<nil>
數到十億：context deadline exceeded true
  • 連線在 Exec 完成、QueryRow 的 Scan 完成、Rows 被關掉、交易 Commit 或 Rollback 之後才會回到池子。忘了任何一個，那條連線就借了不還。
  • db.Stats() 回傳連線池的狀態：使用中幾條、閒置幾條、累計等了幾次、等了多久。放進監控，連線外洩在變成事故之前就看得到。
  • context 的期限到了，database/sql 會取消正在進行的查詢，modernc.org/sqlite 會中斷 SQLite 的執行。HTTP 服務裡把 r.Context() 一路傳下來，客人斷線時查詢也會停（26 章）。
  • 其他設定：SetMaxIdleConns（池子裡最多留幾條閒置的）、SetConnMaxIdleTime 和 SetConnMaxLifetime（多久換一條新的；資料庫或中間的防火牆會切斷太久的連線）。

8. migration 與測試

資料表要在程式第一次執行時建立，之後每次改結構（加欄位、加索引）都要套用到所有已經存在的資料庫上。這就是 migration。SQL 檔放在 migrations 目錄：

-- 菜單：每道菜的名稱不能重複，價格要大於 0，庫存不能是負的。
CREATE TABLE dishes (
    id    INTEGER PRIMARY KEY,
    name  TEXT    NOT NULL UNIQUE,
    price INTEGER NOT NULL CHECK (price > 0),
    stock INTEGER NOT NULL DEFAULT 0 CHECK (stock >= 0)
);
-- 訂單：dish_id 一定要對到一道存在的菜；備註可以是 NULL。
CREATE TABLE orders (
    id      INTEGER PRIMARY KEY,
    dish_id INTEGER NOT NULL REFERENCES dishes (id),
    qty     INTEGER NOT NULL CHECK (qty > 0),
    total   INTEGER NOT NULL,
    note    TEXT
);

CREATE INDEX orders_dish_id ON orders (dish_id);

用 20 章的 embed 把它們編進執行檔，部署時就不用另外帶檔案。已經跑到第幾個，記在 SQLite 資料庫檔案裡的 PRAGMA user_version：

var migrations embed.FS

func migrate(ctx context.Context, db *sql.DB) error {
	names, err := fs.Glob(migrations, "migrations/*.sql")
	if err != nil {
		return err
	}
	var done int
	if err := db.QueryRowContext(ctx, "PRAGMA user_version").Scan(&done); err != nil {
		return err
	}
	if done > len(names) {
		return fmt.Errorf("資料庫的版本 %d 比程式認得的 %d 新，請更新程式", done, len(names))
	}
	for i := done; i < len(names); i++ {
		if err := applyMigration(ctx, db, names[i], i+1); err != nil {
			return fmt.Errorf("%s：%w", names[i], err)
		}
	}
	return nil
}

func applyMigration(ctx context.Context, db *sql.DB, name string, version int) error {
	stmts, err := migrations.ReadFile(name)
	if err != nil {
		return err
	}
	tx, err := db.BeginTx(ctx, nil)
	if err != nil {
		return err
	}
	defer tx.Rollback() // Commit 成功之後再 Rollback 不會有作用

	if _, err := tx.ExecContext(ctx, string(stmts)); err != nil {
		return err
	}
	// PRAGMA 不能用 ? 帶參數；version 是程式自己算出來的整數，不是外面來的輸入，所以可以直接放進字串。
	if _, err := tx.ExecContext(ctx, fmt.Sprintf("PRAGMA user_version = %d", version)); err != nil {
		return err
	}
	return tx.Commit()
}
func migrateDemo() {
	ctx := context.Background()
	dir, err := os.MkdirTemp("", "bento-*")
	if err != nil {
		fmt.Println(err)
		return
	}
	defer os.RemoveAll(dir)
	path := filepath.Join(dir, "bento.db")

	for i := 1; i <= 2; i++ {
		db, err := openDB(ctx, path)
		if err != nil {
			fmt.Println(err)
			return
		}
		var version int
		db.QueryRowContext(ctx, "PRAGMA user_version").Scan(&version)
		fmt.Printf("第 %d 次開啟：版本 %d\n", i, version)

			// 假裝有一個更新版的程式，已經把這個資料庫升到第 3 版。
			db.ExecContext(ctx, "PRAGMA user_version = 3")
		}
		db.Close()
	}

	_, err = openDB(ctx, path)
	fmt.Printf("舊程式開新資料庫：%v\n", err)
}
第 1 次開啟：版本 2
第 2 次開啟：版本 2
舊程式開新資料庫：資料庫的版本 3 比程式認得的 2 新，請更新程式
  • 每個檔案和版本號在同一個交易裡：SQLite 的 CREATE TABLE 也能復原，中途失敗不會留下一半的結構。MySQL 的 DDL 會自動 Commit，沒有這個保證。
  • 舊版的程式遇到更新的資料庫時拒絕開啟，避免它用舊的認知寫入資料。
  • 其他資料庫沒有 user_version，常見的做法是建一個 schema_migrations 資料表記錄版本。社群也有現成的 migration 工具，需要往回復原（down migration）、多人同時開發時可以考慮。

測試直接用真的資料庫。SQLite 只是一個檔案，每個測試在 t.TempDir() 開一個新的，測試結束自動刪除；第 6 節的 newTestStore 就是這樣做的。這比用假的 store 測試更有價值：SQL 寫錯、限制、交易的行為，假的都測不到。

要小心的是 ":memory:"：記憶體裡的資料庫只屬於開它的那一條連線，連線池裡的另一條連線看到的是另一個空的資料庫：

func TestMemoryPerConnection(t *testing.T) {
	ctx := t.Context()
	db, err := sql.Open("sqlite", ":memory:")
	if err != nil {
		t.Fatal(err)
	}
	defer db.Close()

	c1, err := db.Conn(ctx)
	if err != nil {
		t.Fatal(err)
	}
	defer c1.Close()
	c2, err := db.Conn(ctx)
	if err != nil {
		t.Fatal(err)
	}
	defer c2.Close()

	if _, err := c1.ExecContext(ctx, `CREATE TABLE dishes (name TEXT)`); err != nil {
		t.Fatal(err)
	}
	_, err = c2.ExecContext(ctx, `INSERT INTO dishes VALUES ('排骨飯')`)
	if err == nil || !strings.Contains(err.Error(), "no such table") {
		t.Errorf("第二條連線：err = %v，want no such table", err)
	}
}
  • 測試偶爾出現「no such table」，常常就是這個原因：查詢剛好拿到了另一條連線。用暫存檔案最單純。
  • 正式環境用 MySQL、PostgreSQL 的話，測試也該用同一種資料庫（例如在 CI 裡用容器跑一個），SQLite 的行為和它們不完全一樣。

常見錯誤

1. 用字串組 SQL

使用者的輸入會變成 SQL 的一部分。值一律用 ? 帶入，欄位名稱用白名單（第 4 節）。

2. rows 沒有關

特別是迴圈中間的 return：連線借了不還，連線池用光後所有查詢都卡住。QueryContext 成功後立刻 defer rows.Close()（第 3、7 節，練習 6）。

3. 沒有檢查 rows.Err()

讀到一半出錯時，Next 一樣回傳 false，不完整的結果會被當成完整的（第 3 節）。

4. 每個請求 sql.Open 一次

*sql.DB 是連線池，整個程式開一個、共用。每次都開，連線池就沒有作用，用完也常常忘了關（第 1 節）。

5. 交易裡用 db 而不是 tx

那個查詢不在交易裡：看不到交易裡的改變，失敗時也不會復原。連線池上限很小時甚至會卡死（第 6 節）。

6. 把可能是 NULL 的欄位 Scan 進 string

平常資料都有值所以沒事，第一筆 NULL 出現時才出錯。用 sql.Null[T]、指標或 COALESCE（第 5 節）。

7. ":memory:" 搭配連線池

每條連線各有一個資料庫，資料表時有時無。測試用 t.TempDir() 裡的檔案（第 8 節）。

8. 使用沒有 Context 的方法

db.Query、db.Exec 不能取消，慢查詢會一直跑下去。一律用 QueryContext、ExecContext，把請求的 ctx 傳進來（第 7 節）。

重點回顧

  • database/sql 是介面，driver 用空白 import 註冊；*sql.DB 是連線池，整個程式開一個。
  • sql.Open 不會連線，開啟後用 PingContext 確認。
  • 寫入用 ExecContext，看 LastInsertId 和 RowsAffected；資料表的限制是最後一道防線。
  • 多列用 QueryContext：defer rows.Close()、Next 迴圈、Scan、rows.Err()；一列用 QueryRowContext，查不到是 sql.ErrNoRows。
  • 值一律用參數帶入，欄位名稱用白名單。
  • 可能是 NULL 的欄位用 sql.Null[T]、指標或 COALESCE。
  • 交易：BeginTx、defer Rollback、所有查詢用 tx、最後 Commit；並發時用鎖或條件式 UPDATE 防止超賣。
  • 連線借了要還；所有的方法用 Context 版本，查詢才能被取消。
  • migration 依編號執行並記錄版本；測試用暫存目錄裡真的資料庫。
